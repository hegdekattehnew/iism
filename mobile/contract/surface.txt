# The API operations the released mobile app calls (Sprint 62, ADR-072).
#
# One per line, "METHOD /path". The CI check `make check-contract` compares each of these against
# the snapshot taken at the last store release and fails on a change that would break an installed
# app. Add an operation here in the same change that makes the app call it, then run
# `make snapshot-contract`; nothing outside this list is checked, because the web app deploys with
# the API and may change freely.

# Health: the app's own start-up probe.
GET /health

# Sign in and out, by phone (candidate-only for v1: an organisation signs in on the web).
POST /auth/otp/request
POST /auth/otp/verify
POST /auth/refresh
POST /auth/logout
POST /auth/logout-all
GET /auth/me

# Browse without signing in (M2, Sprint 63): vacancies, courses, standards, and the search over them.
GET /jobs
GET /jobs/{slug}
GET /courses
GET /courses/{slug}
GET /skills
GET /skills/{slug}
GET /skills/search
GET /skills/{slug}/jobs
GET /skills/{slug}/courses
GET /roles/search

# The API operations the released mobile app calls (Sprint 62, ADR-072).
#
# One per line, "METHOD /path". The CI check `make check-contract` compares each of these against
# the snapshot taken at the last store release and fails on a change that would break an installed
# app. Add an operation here in the same change that makes the app call it, then run
# `make snapshot-contract`; nothing outside this list is checked, because the web app deploys with
# the API and may change freely.

# Health: the app's own start-up probe.
GET /health

# Sign in and out, by phone (candidate-only for v1: an organisation signs in on the web).
POST /auth/otp/request
POST /auth/otp/verify
POST /auth/refresh
POST /auth/logout
POST /auth/logout-all
GET /auth/me

# Browse without signing in (M2, Sprint 63): vacancies, courses, standards, and the search over them.
GET /jobs
GET /jobs/{slug}
GET /courses
GET /courses/{slug}
GET /skills
GET /skills/{slug}
GET /skills/search
GET /skills/{slug}/jobs
GET /skills/{slug}/courses
GET /roles/search

# The candidate's own side (M3, Sprint 64): profile and skills, matches, applying, saving a vacancy
# and registering interest in a course.
GET /me/profile
PUT /me/profile
POST /me/profile/skills/bulk
DELETE /me/profile/skills/{skill_slug}
GET /roles/{slug}/standards
GET /me/matches
GET /me/matches/{slug}
GET /me/applications
POST /me/applications
POST /me/applications/{application_id}/withdraw
GET /me/saved-jobs
POST /me/saved-jobs
DELETE /me/saved-jobs/{job_slug}
GET /me/course-interests
POST /me/course-interests
POST /me/course-interests/{interest_id}/withdraw

# Notices, push, account and a way to complain (M4, Sprint 65): the updates a person is told, the
# phones that may be sent a push, taking their data or leaving, a message to the team, and
# reporting a listing.
GET /me/notifications
GET /me/notifications/unread-count
POST /me/notifications/read
POST /me/notifications/{notification_id}/read
POST /me/devices
POST /me/devices/unregister
GET /me/account/export
GET /me/account/deletion
DELETE /me/account
POST /grievances
GET /me/grievances
POST /jobs/{slug}/report
POST /courses/{slug}/report
